```go
// internal/agent/orchestrator.go

type NormalizationContext struct {
    Materials map[string]*MaterialItem  // 已生成的材料清单
    Tools     map[string]*ToolItem      // 已生成的工具清单
    SynonymMap map[string]string         // synonym -> canonical_name
}

// 运行流水线时的步骤
func (m *Orchestrator) Run(ctx context.Context, input *PipelineInput) (*Card, error) {
    // Step 1: 清洗
    cleanResult := m.runCleanAgent(ctx, input.RawText)
    
    // Step 2: 材料/工具（序列运行，不并行）
    materials := m.runMaterialAgent(ctx, cleanResult.CleanedText)
    tools := m.runToolAgent(ctx, cleanResult.CleanedText)
    
    // Step 3: 构建同义词映射表
    normCtx := &NormalizationContext{
        Materials: m.buildMaterialMap(materials),
        Tools:     m.buildToolMap(tools),
        SynonymMap: m.loadSynonymMap(ctx),  // 从数据库加载
    }
    
    // Step 4: 并行运行步骤/风险，但在编排器内做归一化
    steps := m.runStepAgent(ctx, cleanResult.CleanedText)
    risks := m.runRiskAgent(ctx, cleanResult.CleanedText)
    
    // Step 5: **关键**：归一化步骤中的 uses[]
    steps = m.normalizeStepUses(steps, normCtx)
    
    // Step 6: 质检（现在 uses[] 已规范化）
    qcResult := m.runQualityAgent(ctx, input.RawText, &Card{
        Materials: materials,
        Tools: tools,
        Steps: steps,
        Risks: risks,
    })
    
    // Step 7: 代码打分
    confidence := m.scoreConfidence(&Card{...}, qcResult)
    
    return card, nil
}

// 关键函数：归一化 uses[] 
func (m *Orchestrator) normalizeStepUses(
    steps []*StepItem, 
    normCtx *NormalizationContext,
) []*StepItem {
    for _, step := range steps {
        for i, use := range step.Uses {
            normalized := m.normalizeEntityName(use, normCtx)
            if normalized != "" {
                step.Uses[i] = normalized  // 改成标准名称
            }
            // 若无法归一化，保留原值（质检时会标记 ungrounded）
        }
    }
    return steps
}

// 实体名归一化
func (m *Orchestrator) normalizeEntityName(
    name string, 
    normCtx *NormalizationContext,
) string {
    // 优先级 1：精确匹配材料/工具清单
    if mat, ok := normCtx.Materials[name]; ok {
        return mat.Name  // 直接返回材料清单里的规范名
    }
    if tool, ok := normCtx.Tools[name]; ok {
        return tool.Name
    }
    
    // 优先级 2：同义词表匹配
    if canonical, ok := normCtx.SynonymMap[name]; ok {
        // 再检查该 canonical 是否在清单里
        if mat, ok := normCtx.Materials[canonical]; ok {
            return mat.Name
        }
        if tool, ok := normCtx.Tools[canonical]; ok {
            return tool.Name
        }
    }
    
    // 优先级 3：模糊匹配（编辑距离 < 2）
    for matName := range normCtx.Materials {
        if m.levenshteinDistance(name, matName) <= 2 {
            return matName
        }
    }
    for toolName := range normCtx.Tools {
        if m.levenshteinDistance(name, toolName) <= 2 {
            return toolName
        }
    }
    
    // 无法匹配 → 保留原值
    return ""  // 返回空，质检时会标记 grounding 问题
}
```

```go
// internal/quality/scorer.go

func scoreGrounding(card *Card, qcResult *QCEvidence) float64 {
    baseScore := 1.0
    
    // 只扣"真实的"grounding 问题
    for _, issue := range qcResult.GroundingIssues {
        if issue.IssueType == "real_missing" {
            baseScore -= 0.03  // 每个真实缺失扣 0.03
        }
        // semantic_drift 类的问题不扣分（系统已处理）
    }
    
    return max(0, baseScore)
}

// quality_flags 生成时也要区分
func GenerateQualityFlags(qcResult *QCEvidence) []string {
    flags := []string{}
    
    for _, issue := range qcResult.GroundingIssues {
        if issue.IssueType == "real_missing" {
            flags.append(fmt.Sprintf("ungrounded:%s", issue.EntityName))
        }
        // semantic_drift 不加入 flags
    }
    
    return flags
}
```

```go
// internal/auth/permissions.go

type SetPermission string

const (
    PermOwner   SetPermission = "owner"   // 导入者，拥有全部权限
    PermEditor  SetPermission = "editor"  // 被授权编辑
    PermViewer  SetPermission = "viewer"  // 被授权查看
)

// 权限检查函数
func (s *Service) CheckSetPermission(
    ctx context.Context,
    userID uint64,
    setID uint64,
    requiredPerm SetPermission,
) error {
    // Step 1: 若是 admin，直接通过
    user := getUserFromContext(ctx)
    if user.Role == "admin" {
        return nil
    }
    
    // Step 2: 查权限表
    perm := db.GetSetPermission(setID, userID)
    if perm == nil || perm.RevokedAt != nil {
        return errors.New("no permission")
    }
    
    // Step 3: 检查权限等级
    if !hasPermission(perm.Role, requiredPerm) {
        return errors.New("insufficient permission")
    }
    
    return nil
}

// 权限等级检查
func hasPermission(userPerm SetPermission, required SetPermission) bool {
    permLevel := map[SetPermission]int{
        PermOwner:  3,
        PermEditor: 2,
        PermViewer: 1,
    }
    return permLevel[userPerm] >= permLevel[required]
}
```

```go
// internal/repository/card_repo.go

func (r *CardRepository) ListCards(
    ctx context.Context,
    userID uint64,
    userRole string,
) ([]*Card, error) {
    query := db.WithContext(ctx)
    
    if userRole == "admin" {
        // admin 看所有卡片
        // 无过滤
    } else if userRole == "editor" {
        // editor 看自己权限内的教程集的卡片
        query = query.Where(
            "tutorial_id IN (SELECT id FROM tutorials WHERE set_id IN (" +
            "SELECT set_id FROM tutorial_set_permissions " +
            "WHERE user_id = ? AND role IN ('owner', 'editor') AND revoked_at IS NULL" +
            "))",
            userID,
        )
    } else if userRole == "viewer" {
        // viewer 只看 published 的非 research-only 卡片
        query = query.Where(
            "status = 'published' AND " +
            "tutorial_id IN (SELECT id FROM tutorials WHERE set_id IN (" +
            "SELECT set_id FROM tutorial_sets WHERE license != 'research-only'" +
            "))",
        )
    }
    
    return query.Find(&cards).Error
}
```

#### 15.3.2 PITR（按时间点恢复）流程

```bash
# scripts/pitr_recovery.sh
# 按指定时间点恢复数据库

#!/bin/bash

set -e

RECOVERY_TIME="$1"  # 例：2026-10-19 14:30:00
DB_NAME="diy_tutorial_agent"
BINLOG_DIR="/var/log/mysql"
BACKUP_DIR="/backup/mysql"

echo "Starting PITR recovery to: $RECOVERY_TIME"

# Step 1: 查找备份文件
LATEST_BACKUP=$(ls -t $BACKUP_DIR/full_*.sql | head -1)
echo "Using backup: $LATEST_BACKUP"

# Step 2: 新建恢复数据库
RECOVERY_DB="${DB_NAME}_recovery_$(date +%s)"
mysql -uroot -p$MYSQL_PASSWORD -e "CREATE DATABASE $RECOVERY_DB"

# Step 3: 从全量备份恢复
mysql -uroot -p$MYSQL_PASSWORD $RECOVERY_DB < $LATEST_BACKUP

# Step 4: 获取备份的时间戳
BACKUP_TIMESTAMP=$(grep "BACKUP_TIME:" $LATEST_BACKUP | awk '{print $NF}')

# Step 5: 应用 binlog 到指定时间点
mysqlbinlog \
    --stop-datetime="$RECOVERY_TIME" \
    $BINLOG_DIR/mysql-bin.* | \
    mysql -uroot -p$MYSQL_PASSWORD $RECOVERY_DB

echo "PITR completed. Recovery DB: $RECOVERY_DB"
echo "Verify data, then:"
echo "  mysqldump -uroot -p $RECOVERY_DB > verified.sql"
echo "  mysql -uroot -p $DB_NAME < verified.sql"
```

```yaml
# MySQL 主从故障转移配置

---
# 主库故障时，自动将从库提升为主

apiVersion: v1
kind: ConfigMap
metadata:
  name: mysql-failover-config
data:
  failover_script: |
    #!/bin/bash
    # 主库检测失败 3 次 → 触发故障转移
    
    # Step 1: 检查主库健康
    if ! mysqladmin -uhealth -p$HEALTH_PASSWORD ping -h $MASTER_HOST; then
        FAILURE_COUNT=$((FAILURE_COUNT + 1))
        if [ $FAILURE_COUNT -ge 3 ]; then
            echo "Master unreachable, starting failover..."
            
            # Step 2: 在从库上停止复制
            ssh $SLAVE_HOST "mysql -uroot -p$MYSQL_PASSWORD -e 'STOP SLAVE;'"
            
            # Step 3: 获取从库的当前 binlog 位置
            BINLOG_FILE=$(ssh $SLAVE_HOST "mysql -uroot -p -e 'SHOW MASTER STATUS\\G' | grep File | awk '{print \$NF}'")
            BINLOG_POS=$(ssh $SLAVE_HOST "mysql -uroot -p -e 'SHOW MASTER STATUS\\G' | grep Position | awk '{print \$NF}'")
            
            # Step 4: 将从库提升为主库
            ssh $SLAVE_HOST "mysql -uroot -p$MYSQL_PASSWORD -e 'RESET MASTER;'"
            
            # Step 5: 更新应用配置（Kubernetes configmap 或 etcd）
            kubectl patch configmap mysql-config -p "{\"data\":{\"host\":\"$SLAVE_HOST\"}}"
            
            # Step 6: 应用重启（让连接池重新连接到新主库）
            kubectl rollout restart deployment/app
            
            echo "Failover completed. New master: $SLAVE_HOST"
        fi
    else
        FAILURE_COUNT=0
    fi
```

```go
// internal/ai/quality_agent.go

// 系统启动时进行准入测试
func (qa *QualityAgent) PreflightCheck(ctx context.Context) error {
    testCases := loadTestCases("test_data/quality_agent_benchmark.json")
    
    passedCount := 0
    for _, tc := range testCases {
        result := qa.Run(ctx, tc.Input)
        
        // 检查是否捕获了所有预期的问题
        expectedCount := len(tc.ExpectedIssues)
        actualCount := len(result.GroundingIssues) + len(result.StepConflicts)
        
        recall := float64(actualCount) / float64(expectedCount)
        if recall >= 0.90 {
            passedCount++
        } else {
            log.Warnf("Test %s failed: recall=%.2f", tc.Name, recall)
        }
    }
    
    passRate := float64(passedCount) / float64(len(testCases))
    
    // 达到 90% 通过率才启动
    if passRate >= 0.90 {
        log.Infof("Quality Agent preflight passed: %.2f%%", passRate*100)
        return nil
    }
    
    return errors.New("Quality Agent accuracy below 90%, refusing to start")
}

// 系统启动时调用
func init() {
    if err := qualityAgent.PreflightCheck(context.Background()); err != nil {
        log.Fatalf("Preflight check failed: %v", err)
        os.Exit(1)
    }
}
```

```go
func (m *QCMonitor) WeeklyAccuracyCheck(ctx context.Context) {

    // 随机抽取 50 张复核过的卡片
    reviewedCards := db.GetCardsByReviewDateRange(lastWeek, time.Now(), 50)

    for _, card := range reviewedCards {

        qcResult := runQualityAgent(ctx, card)
        
        // 与复核员的修改对比
        originalQCResult := card.QualityFlags  // 原始质检结果

                if qcIssue.IssueType == reviewChange.IssueType &&

                    found = true
                    break
                }
            }

                matchCount++
            } else {
                log.Warnf("QC Agent missed issue: %v", reviewChange)
                
                // 记录漏判
                metrics.IncrementQCMissCount(reviewChange.IssueType)
            }
        }
    }
    
    actualRecall := float64(matchCount) / float64(len(reviewedCards))
    
    // 与 SLA 目标对比（90%）
    if actualRecall < 0.90 {
        sendAlert(fmt.Sprintf(
            "Quality Agent recall dropped to %.2f%% (target: 90%%)",
            actualRecall*100,
        ))
        
        // 自动降级：新投递的任务全部进人工复核

    }
```

## 15.5 缺陷 3 修复：卡片版本化的并发安全方案

### 问题回顾
重跑时 `is_current` 的唯一性保证有竞态条件。

### 修复方案：分布式锁 + 明确恢复策略

#### 15.5.1 重跑锁机制

```go
// internal/service/pipeline_service.go

func (s *PipelineService) RestructurizeCard(
    ctx context.Context,
    tutorialID uint64,
    req *RestructureRequest,
) (*Card, error) {
    // Step 1: **获取分布式锁**，防止并发重跑
    lockKey := fmt.Sprintf("tutorial:rerun:%d", tutorialID)
    lock := s.redis.SetNX(lockKey, "locked", 30*time.Second)
    
    if !lock {
        return nil, errors.New("tutorial is being rerun, please wait")
    }
    defer s.redis.Del(lockKey)
    
    // Step 2: 获取旧版本卡片
    oldCard := db.GetCurrentCard(tutorialID)
    
    // Step 3: 运行编排器生成新版本
    newCard := s.orchestrator.Run(ctx, &PipelineInput{
        RawText: req.RawText,
    })
    
    // Step 4: 在事务内更新版本
    return db.WithTx(func(tx *gorm.DB) (*Card, error) {
        // 4a. 旧版本标记为 is_current=0
        if oldCard != nil {
            tx.Model(oldCard).Update("is_current", false)
        }
        
        // 4b. 新版本标记为 is_current=1
        newCard.CardVersion = (oldCard.CardVersion if oldCard != nil else 0) + 1
        newCard.IsCurrent = true
        
        // 4c. 插入新版本
        if err := tx.Create(newCard).Error; err != nil {
            return nil, err
        }
        
        return newCard, nil
    })
}
```
