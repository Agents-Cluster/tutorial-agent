## 9.3 初始化脚本 scripts/init.sh

> 在**宿主机**执行，通过 `docker exec` 进入容器操作。顺序不能乱：先启动所有容器 → 等待 healthcheck → 执行 init 脚本 → 启动 Go 服务。

```bash
#!/bin/bash
set -euo pipefail

# 0. 变量从 .env 读，与 compose 同源；脚本内不写任何密码（否则改密码后脚本必挂，也违反 7.7 第 2 项）
#    .env.root 只含 MYSQL_ROOT_PASSWORD，单独一个文件是为了"root 密码不进应用容器"
set -a
[ -f ./.env ] && . ./.env
[ -f ./scripts/.env.root ] && . ./scripts/.env.root
set +a
: "${MYSQL_ROOT_PASSWORD:?MYSQL_ROOT_PASSWORD 未设置，请先按 9.5 第 2 步准备 scripts/.env.root}"
: "${MYSQL_USER:?MYSQL_USER 未设置（应用账号，见 9.2）}"
: "${MYSQL_PWD:?MYSQL_PWD 未设置（应用账号密码，见 9.2）}"
: "${MINIO_ACCESS_KEY:?MINIO_ACCESS_KEY 未设置}"

echo "===== 开始初始化资源 ====="

# 1. 初始化 MySQL 表结构（DDL 只由 root 执行）
echo "初始化 MySQL 表结构..."
docker exec -i diy_mysql mysql -uroot -p"$MYSQL_ROOT_PASSWORD" "$MYSQL_DATABASE" < ./scripts/init.sql

# 1.1 创建应用账号并只授 DML 权限（7.6 最小权限；应用被注入也改不了表结构）
#     必须在建表之后执行：MySQL 的库级 GRANT 对"之后创建的表"同样生效，但先建库更直观
echo "创建应用账号 ${MYSQL_USER}（仅 DML）..."
docker exec -i diy_mysql mysql -uroot -p"$MYSQL_ROOT_PASSWORD" <<SQL
CREATE USER IF NOT EXISTS '${MYSQL_USER}'@'%' IDENTIFIED BY '${MYSQL_PWD}';
ALTER USER '${MYSQL_USER}'@'%' IDENTIFIED BY '${MYSQL_PWD}';
GRANT SELECT, INSERT, UPDATE, DELETE ON \`${MYSQL_DATABASE}\`.* TO '${MYSQL_USER}'@'%';
FLUSH PRIVILEGES;
SQL
# 注：新账号默认只有 USAGE，不需要 REVOKE；`CREATE USER IF NOT EXISTS` + `ALTER USER` 保证脚本可重复执行

# 2. 初始化 RabbitMQ 交换机、队列、死信队列（DLX）
echo "初始化 RabbitMQ 队列与 DLQ..."

# 2.1 业务交换机
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare exchange name=agent_task_exchange type=direct

# 2.2 死信交换机 + 死信队列
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare exchange name=agent_dlx_exchange type=direct
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare queue name=agent_dlq_queue
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" bind queue=agent_dlq_queue exchange=agent_dlx_exchange routing_key=dlq

# 2.3 业务队列（P0/P1/P2），都配置死信转发到 dlx_exchange
#     arguments: x-dead-letter-exchange=agent_dlx_exchange, x-dead-letter-routing-key=dlq
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare queue name=agent_p0_queue \
  arguments='{"x-dead-letter-exchange":"agent_dlx_exchange","x-dead-letter-routing-key":"dlq"}'
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare queue name=agent_p1_queue \
  arguments='{"x-dead-letter-exchange":"agent_dlx_exchange","x-dead-letter-routing-key":"dlq"}'
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" declare queue name=agent_p2_queue \
  arguments='{"x-dead-letter-exchange":"agent_dlx_exchange","x-dead-letter-routing-key":"dlq"}'

# 2.4 业务队列绑定到业务交换机
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" bind queue=agent_p0_queue exchange=agent_task_exchange routing_key=p0
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" bind queue=agent_p1_queue exchange=agent_task_exchange routing_key=p1
docker exec diy_rabbitmq rabbitmqadmin -u "$RABBITMQ_USER" -p "$RABBITMQ_PASS" bind queue=agent_p2_queue exchange=agent_task_exchange routing_key=p2

# 3. 初始化 MinIO 存储桶
#    用一次性 mc 容器、复用 minio 的网络命名空间：minio 服务镜像不保证自带 mc，
#    exec 进去跑 mc 在部分版本上直接 command not found（与 9.9.1 备份脚本同一套写法）
echo "创建 MinIO 桶..."
MC_HOST="http://${MINIO_ACCESS_KEY}:${MINIO_SECRET_KEY}@127.0.0.1:9000"
docker run --rm --network "container:diy_minio" -e MC_HOST_local="$MC_HOST" \
  minio/mc:latest mc mb --ignore-existing "local/${MINIO_BUCKET}"
# mc 新版本已弃用 `mc policy`，等价写法是 `mc anonymous`
docker run --rm --network "container:diy_minio" -e MC_HOST_local="$MC_HOST" \
  minio/mc:latest mc anonymous set none "local/${MINIO_BUCKET}"

# 3.1 备份专用桶：9.9.1 的备份脚本往 local/${MINIO_BACKUP_BUCKET}/mysql/ 上传，
#     这里不建，备份第一次执行就报 Bucket does not exist（很容易被误判成 mc 用法问题）
docker run --rm --network "container:diy_minio" -e MC_HOST_local="$MC_HOST" \
  minio/mc:latest mc mb --ignore-existing "local/${MINIO_BACKUP_BUCKET}"
docker run --rm --network "container:diy_minio" -e MC_HOST_local="$MC_HOST" \
  minio/mc:latest mc anonymous set none "local/${MINIO_BACKUP_BUCKET}"

echo "===== 初始化全部完成 ====="
```

## 9.4 docker-compose.yml 完整配置

> 放在项目根目录，固化所有组件端口、镜像、依赖顺序。

```yaml
services:
  mysql:
    image: mysql:8.0.36
    container_name: diy_mysql
    # 与 3.4.4（UTC）、9.9（binlog for PITR）保持一致
    command: >
      --log-bin=/var/lib/mysql/binlog
      --binlog-format=ROW
      --binlog-expire-logs-seconds=604800
      --server-id=1
      --default-time-zone=+00:00
      --character-set-server=utf8mb4
      --collation-server=utf8mb4_unicode_ci
    environment:
      MYSQL_DATABASE: ${MYSQL_DATABASE:-diy_tutorial_agent}
    # root 密码单独一个 env_file（MYSQL_ROOT_PASSWORD）：既是 mysql 镜像要求的变量名，
    # 也避免它出现在 app 容器的环境里（app 的 env_file 只有 .env）
    env_file:
      - ./scripts/.env.root
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql
    healthcheck:
      # $$ 是 compose 的转义：容器内展开 $MYSQL_ROOT_PASSWORD，不在这里写明文
      test: ["CMD-SHELL", "mysqladmin ping -uroot -p\"$$MYSQL_ROOT_PASSWORD\""]
      interval: 10s
      timeout: 5s

  redis:
    image: redis:7.2-alpine
    container_name: diy_redis
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5

  rabbitmq:
    image: rabbitmq:3.13-management
    container_name: diy_rabbitmq
    environment:
      RABBITMQ_DEFAULT_USER: ${RABBITMQ_USER:-admin}
      RABBITMQ_DEFAULT_PASS: ${RABBITMQ_PASS:-admin123}

  minio:
    image: minio/minio:RELEASE.2024-05-01T01-11-41Z

    container_name: diy_prom

      - "9090:9090"

      - ./server/deploy/prometheus.yml:/etc/prometheus/prometheus.yml

    image: grafana/grafana:10.4.0
    container_name: diy_grafana
    ports:
      - "3000:3000"
    volumes:

  app:
    build:
      # 构建上下文必须是**仓库根**：app 启动时要读 docs/prompts/*.md（02:198、03:281），
      # 用 ./server 作上下文的话 docs/ 根本进不了构建缓存，镜像里没有 Prompt，容器启动即自杀
      context: .
      dockerfile: server/deploy/Dockerfile
    container_name: diy_app
    ports:
      - "8080:8080"
    env_file: .env
    depends_on:
      mysql:
        condition: service_healthy
      redis:
        condition: service_healthy
      rabbitmq:
        condition: service_healthy
      minio:
        condition: service_healthy
    # 健康检查打根路径 /health（不加 /api/v1，见 4.1）；curl 由 9.4.2 的 Dockerfile 装上
    healthcheck:
      test: ["CMD", "curl", "-fsS", "http://127.0.0.1:8080/health"]
      interval: 10s
      timeout: 5s
      retries: 5

      timeout: 3s

  mysql_data:
  redis_data:
  rabbitmq_data:
  minio_data:
  prom_data:
  grafana_data:
```

> **healthcheck 的两个坑**：
> 1. `minio` 镜像部分版本不含 `curl`，若 `docker compose ps` 显示 `unhealthy` 但服务正常，改用
>    `test: ["CMD-SHELL", "exec 3<>/dev/tcp/127.0.0.1/9000 && echo ok"]`，或先 `mc alias set` 后用 `mc ready local`；
> 2. MySQL 的 healthcheck 用 `mysqladmin ping` 只证明"进程活着"，不证明"库可用"。初始化脚本执行前，
>    额外跑一次 `SELECT 1` 更稳（`mysql -e "SELECT 1"`）。

### 9.4.2 后端 Dockerfile：Prompt 必须打进镜像

`prompts.go` 是按 `PROMPT_DIR` 读文件的（02:198），而 Prompt 正文在仓库根的 `docs/prompts/`。构建上下文改成仓库根之后（9.4），Dockerfile 必须显式把它 COPY 进去，并在**构建期**校验数量：

```dockerfile
# server/deploy/Dockerfile（构建上下文 = 仓库根）
FROM golang:1.22-alpine AS build
WORKDIR /src
COPY server/go.mod server/go.sum ./
RUN go mod download
COPY server/ ./
RUN CGO_ENABLED=0 go build -trimpath -o /out/app ./cmd/server

FROM alpine:3.19
# curl 供 compose 的 healthcheck 使用（裸 alpine 没有；不装就换成 wget，别两个都不装）
RUN apk add --no-cache ca-certificates tzdata curl && adduser -D -u 10001 app
WORKDIR /app
COPY --from=build /out/app ./app
# 关键一行：Prompt 与二进制同镜像；漏了它，容器启动时加载失败直接退出
COPY docs/prompts/ ./docs/prompts/
# 构建期校验：必须恰好 6 个 .md（与 README 的 Prompt 库一致），少一个就让 build 失败
RUN test "$(ls -1 ./docs/prompts/*.md | wc -l)" -eq 6 \
    || (echo "docs/prompts 必须恰好 6 个 .md" && exit 1)
USER app
ENV PROMPT_DIR=/app/docs/prompts
EXPOSE 8080
ENTRYPOINT ["./app"]
```

四条约束：
1. **`COPY docs/prompts/` 不能省**：build 阶段只产二进制的话，镜像里没有 Prompt，启动自检（3.2 注释）直接退出——报错写的是"Prompt 缺失"，很容易被误判成代码 bug；
2. **根目录 `.dockerignore` 不得排除 `docs/`**：否则 COPY 出来的目录是空的，第 3 步的数量校验会拦下；
3. **数量校验放在 `docker build` 里**：问题从"运行时报错"前移到"构建失败"，CI 里就能看见；
4. `PROMPT_DIR` 在镜像内固定 `/app/docs/prompts`（9.2），本地 `go run` 时覆盖为 `./docs/prompts`。

## 9.5 部署执行步骤（固定顺序）

> 严格按顺序执行，防止时序 bug（时序问题是最常见坑）。

1. 前置检查：执行环境检查，确认端口、磁盘、网络；
2. 复制 `.env.example` 为 `.env`，填入 LLM API Key、修改默认密码；
3. 拉取基础镜像并构建应用镜像：`docker compose pull` + `docker compose build --pull`
   （`app` / `web` 是 `build` 型服务，只 `pull` 不会更新它们，这是最常被漏掉的一步）；
4. 启动全部依赖容器：`docker compose up -d mysql redis rabbitmq minio prometheus grafana`；
5. **等待所有容器 healthcheck 全部通过**（非常关键，否则初始化会连接失败）；
6. 执行初始化脚本 `bash scripts/init.sh`：创建数据库、数据表、MQ 队列、MinIO 桶；
7. 启动后端服务 `docker compose up -d app`；
8. 启动前端 `docker compose up -d web`；
9. 执行系统健康校验脚本 `bash scripts/health_check.sh`；
10. 执行冒烟测试，跑通完整主流程：导入教程集 → 提交任务 → 等待流水线 → 生成卡片 → 复核低置信 → 导出。

## 9.6 系统健康校验脚本

> 部署完成后自动校验所有依赖连通性，提前发现连接类 bug。保存为 `scripts/health_check.sh`。

```bash
#!/bin/bash
set -euo pipefail
set -a; [ -f ./.env ] && . ./.env; set +a

echo "===== 健康检查开始 ====="

# 1. HTTP 服务健康检查
curl -s http://127.0.0.1:8080/health || { echo "❌ app服务异常"; exit 1; }
echo "✅ Go服务健康检查通过"

# 2. MySQL 连通（用 SELECT 1 而不是 ping：ping 只证明进程活着，不证明库可用）
#    用**应用账号**探活，不用 root：①root 密码不在 .env 里（7.6 最小权限），②应用视角能连通才算健康
docker exec diy_mysql mysql -u"$MYSQL_USER" -p"$MYSQL_PWD" -e "SELECT 1" "$MYSQL_DATABASE" > /dev/null \
  && echo "✅ MySQL正常"

# 3. Redis 连通
docker exec diy_redis redis-cli ping | grep -q PONG && echo "✅ Redis正常"

# 4. RabbitMQ 连通
docker exec diy_rabbitmq rabbitmqctl status | grep -q "Runtime node" && echo "✅ RabbitMQ正常"

# 5. MinIO 连通
curl -sf http://127.0.0.1:9000/minio/health/live && echo "✅ MinIO正常"

# 6. 前端容器可达（打 nginx 自己的探活端点，不经过后端）
curl -sf http://127.0.0.1:3001/healthz > /dev/null && echo "✅ 前端容器正常"

# 7. 反代链路可达（穿过 nginx 打到 app 的 /health，验证 9.12 的同源转发真的通）
#    只测第 6 步的话，nginx 起来了但 proxy_pass 主机名写错也能"通过"
curl -sf http://127.0.0.1:3001/health > /dev/null && echo "✅ 同源反代链路正常"

echo "===== 全部健康检查通过，可以开始冒烟测试 ====="
```

## 9.7 冒烟测试用例清单

> 部署完成必跑，验证核心链路端到端无 bug。

1. 账号登录：管理员账号登录，获取 token；
2. 教程集导入：上传预处理后的 DIY 教程样本，校验元数据入库；

6. 校验卡片：清洗/材料/工具/步骤/风险/质检全流水线跑通，卡片 JSON 字段完整，符合 Card Schema；

10. 数据导出：导出标准 JSON/CSV，验证文件可下载；

13. 权限校验：editor 无法访问 stats 接口，viewer 无法提交任务；

## 9.8 故障排查手册

> 出现报错快速定位。

| 现象 | 根因 | 修复方案 |
| --- | --- | --- |
| app 启动报错：连接 mysql 拒绝 | MySQL 容器未就绪 | 等待 healthcheck 通过再执行初始化 |
| MQ 消息重复消费 | ACK 异常，缺少幂等 | 确认消费幂等逻辑，成功后写 Redis 标记（TTL 7天） |
| 任务提交后一直 pending 不执行 | outbox 投递协程异常 / MQ 不可用 | 检查 outbox 表 status=pending 记录，手动触发重投 |
| LLM 输出 JSON 解析失败率高 | Prompt 约束不足 | 检查 Agent Prompt 的 JSON Schema 约束，启用重试 |
| 步骤顺序频繁错乱 | 步骤 Agent 输出不稳定 | 检查步骤 Prompt 与质检时序冲突对逻辑 |
| LLM 接口调用失败 | API 密钥错误/网络不通 | 检查 .env 的 LLM_API_KEY、宿主机网络连通性 |
| 任务一直 running 不结束 | LLM 调用超时未熔断 | 检查超时配置与熔断阈值 |
| MinIO 上传 403 | bucket 不存在/密钥错误 | 重新执行 init 脚本创建桶 |
| 低置信样本堆积复核队列 | 置信度阈值设置过高 | 调整 QUALITY_CONFIDENCE_THRESHOLD |
| 同一篇教程重跑后卡片没变 | 唯一键冲突 / is_current 未更新 | 检查 cards 表 uk_tutorial_version 与 is_current 逻辑 |

## 9.9 备份与恢复

> 8.2 承诺了 RTO / RPO，这一节给出达成手段。**没有这一节，RPO ≤ 1 分钟就是空话。**

### 9.9.1 备份策略

| 对象 | 方式 | 频率 | 保留 | 存放 |
| --- | --- | --- | --- | --- |
| MySQL 全量 | `mysqldump --single-transaction --routines` | 每日 03:00 | 7 天 | MinIO `diy-backup` 桶的 `mysql/` 前缀（`MINIO_BACKUP_BUCKET`，由 init.sh 创建）+ 本地一份 |
| MySQL 增量 | binlog（`log_bin=ON`, `binlog_format=ROW`） | 实时 | 7 天 | 宿主机卷 + 每日同步到 MinIO `diy-backup/binlog/` |
| MinIO 原始文件 | 不备份 | — | — | 源数据可重放，卡片可重建 |
| Redis | 不备份 | — | — | 只存锁/限流/幂等标记，可重建 |
| 配置 | `.env` + `configs/` | 每次变更 | 全量 | 私有 Git 仓库（不含密钥，密钥走密钥管理） |

**备份脚本**（`scripts/backup.sh`，挂 cron）：

```bash
#!/bin/bash
set -euo pipefail
# 两个 env 都要 source：.env 给库名与 MinIO 凭据，.env.root 给 root 密码——
# mysqldump 的 --routines/--triggers 需要超出 app 账号 DML 的权限（7.6）。
# 此前只 source .env 却用 -uroot -p"$MYSQL_PWD"，而 MYSQL_PWD 是 app 账号密码，必然认证失败。
set -a; [ -f ./.env ] && . ./.env; [ -f ./scripts/.env.root ] && . ./scripts/.env.root; set +a
: "${MYSQL_ROOT_PASSWORD:?MYSQL_ROOT_PASSWORD 未设置，见 9.3}"
STAMP=$(date -u +%Y%m%dT%H%M%SZ)
BACKUP_DIR="${BACKUP_DIR:-/backup/mysql}"
mkdir -p "$BACKUP_DIR"

# 1) 全量导出到宿主机（--single-transaction 保证不锁表）
docker exec diy_mysql mysqldump -uroot -p"$MYSQL_ROOT_PASSWORD" \
  --single-transaction --routines --triggers "$MYSQL_DATABASE" \
  | gzip > "${BACKUP_DIR}/${STAMP}.sql.gz"

# 2) 上传到 MinIO：用一次性 mc 容器，复用 minio 的网络命名空间
#    不要把 mc 跑在 diy_minio 容器里——那个容器没有宿主机的 /backup 目录
docker run --rm --network "container:diy_minio" \
  -v "${BACKUP_DIR}:/backup:ro" \
  -e MC_HOST_local="http://${MINIO_ACCESS_KEY}:${MINIO_SECRET_KEY}@127.0.0.1:9000" \
  minio/mc:latest mc cp "/backup/${STAMP}.sql.gz" "local/${MINIO_BACKUP_BUCKET}/mysql/"

# 3) 本地保留 7 天
find "$BACKUP_DIR" -name '*.sql.gz' -mtime +7 -delete
```

> 前置条件：MySQL 需开启 binlog（`--log-bin --binlog-format=ROW --binlog-expire-logs-seconds=604800`），
> 否则只有全量、没有 PITR，A3 的 RPO 目标不成立（见 11.1.2）。

### 9.9.2 恢复步骤（按时间点）

## 9.11 定时任务（宿主机 crontab）

> compose 里**故意不放调度容器**：单机部署下多一个 cron 容器就多一个要监控的进程，而所有脚本都在仓库根的 `scripts/` 下（与 02 目录树、9.3/9.6/9.9 的路径口径一致），宿主机 crontab 直接调即可。

| 脚本 | 频率 | 作用 | 出处 |
| --- | --- | --- | --- |
| `scripts/backup.sh` | 每日 03:00 | MySQL 全量 + 上传 `diy-backup` 桶 + 本地保留 7 天 | 9.9.1 |
| `scripts/integrity_check.sh` | 每日 05:00 | 扫孤儿记录（`cards.tutorial_id`、`card_entities.card_id` 不存在，`export_tasks.file_key` 对应对象缺失）并输出报告 | 3.4.1 |
| `scripts/archive.sh` | 每月 1 日 04:00 | 把 `agent_items` / `operation_logs` 中 `updated_at < 上月 1 日` 的终态行迁到 `*_archive`（脚本内 `CREATE TABLE IF NOT EXISTS ... LIKE` 建表），并删 `export_tasks` 的 `expired` 行 | 3.4.5 |

```cron
# crontab -e（宿主机；工作目录必须是仓库根，脚本用相对路径读 .env 与 scripts/.env.root）
# 首次安装先建日志目录：logs/ 不存在时 >> 重定向会失败，cron 不报错、任务却静默不执行
mkdir -p logs

0 3 * * *  cd /opt/diy-tutorial-agent && bash scripts/backup.sh          >> logs/cron.log 2>&1
5 5 * * *  cd /opt/diy-tutorial-agent && bash scripts/integrity_check.sh >> logs/cron.log 2>&1
0 4 1 * *  cd /opt/diy-tutorial-agent && bash scripts/archive.sh         >> logs/cron.log 2>&1
```

三条约定：
1. **outbox 兜底重投不进 crontab**：它由应用内协程做（6.3 第 4 条），放进 cron 会与应用协程抢同一批行；两者都用 `SELECT ... FOR UPDATE SKIP LOCKED`，即使重复触发也不会重复投递；
2. 三个脚本都必须**幂等可重跑**（`mc mb --ignore-existing`、`CREATE TABLE IF NOT EXISTS`、`mysqldump` 覆盖同名文件），手工补跑不会造成二次副作用；
3. `logs/cron.log` 已在 `.gitignore` 的 `logs/` 规则内（9.10.1），不入库。

## 9.12 前端容器：nginx 静态服务 + `/api` 同源反代

> 此前 compose 里 `web` 只有一个 `3001:80` 的端口映射，**浏览器怎么找到 API 从未定义**：既没有 nginx 配置，也没有构建期注入 API 地址。两种做法（同源反代 / 前端直连 8080 靠 CORS）必须选一种写死，否则前端第一行 `fetch` 就不知道写什么。
>
> **选同源反代**，三个理由：① 8080 不必对外暴露（07.6 端口收敛）；② 浏览器同源，无需处理 CORS 与预检；③ 前端代码用相对路径 `/api/v1/...`，dev/prod 不需要两套 base URL。

### 9.12.1 `web/nginx.conf`

```nginx
server {
    listen 80;
    server_name _;

    # 静态产物（Vite build 输出）
    root /usr/share/nginx/html;
    index index.html;

    # 容器探活端点（compose healthcheck 与 9.6 都打它，不经过后端）
    location = /healthz { return 200 "ok\n"; add_header Content-Type text/plain; }

    # 让健康检查能穿过反代验证整条链（9.6 第 7 步）：/api 之外唯一的代理路径
    location = /health { proxy_pass http://app:8080/health; }

    # 同源反代：浏览器只与 3001 通信，/api 转到 app 容器
    location /api/ {
        proxy_pass http://app:8080;
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        # 批量任务提交是 202 秒回，但导出/统计可能慢：给足读超时，别让 nginx 先 504
        proxy_connect_timeout 5s;
        proxy_read_timeout    60s;
        # 单 upstream，不做健康摘流（摘流口径见 8.2.1，不在这层）
        proxy_next_upstream off;
    }

    # SPA 前端路由回退
    location / { try_files $uri $uri/ /index.html; }
}
```

### 9.12.2 `web/Dockerfile`

```dockerfile
FROM node:20-alpine AS build
WORKDIR /web
COPY package.json package-lock.json ./
RUN npm ci
COPY . .
RUN npm run build            # 产物在 dist/；前端一律用相对路径 /api/v1，不需要注入 base URL

FROM nginx:1.25-alpine
COPY nginx.conf /etc/nginx/conf.d/default.conf
COPY --from=build /web/dist /usr/share/nginx/html
EXPOSE 80
```

### 9.12.3 四条口径（避免与既有设计打架）

1. **这不是 8.2.1 说的"反代"**：8.2.1 与 11.1 的 A2 讲的是 **API 服务前面没有 LB 做健康摘流**（所以摘流靠 `restart: unless-stopped` + 外部探活消费 503）。这里的 nginx 是**前端静态服务器 + 单 upstream 同源转发**，`proxy_next_upstream off`、不做健康检查、不摘流，两者不冲突；
2. **CORS 中间件（02 的 `middleware/cors.go`）只为本地开发存在**：`vite dev` 跑在 5173、直连 8080 时才需要跨源；容器化路径同源，不触发 CORS。白名单只放 `http://localhost:5173` 与 `http://localhost:3001`，不要用 `*`（07.6）；
3. **大文件不走 nginx**：上传用 `POST /uploads` 拿预签名 URL 后**浏览器直传 MinIO**，下载用 `GET /exports/{id}/download` 的 302 直连 MinIO（4.3.6）。所以 `client_max_body_size` 保持默认 1m 即可；**如果哪天改成应用中转，必须同时放大这个值**，否则表现为"上传 413、后端日志一片干净"；
4. **MinIO 9000 必须对浏览器可达**（`MINIO_PUBLIC_ENDPOINT`），这是预签名方案的硬要求，已写进 07.6 的端口收敛；9001 控制台不暴露。

***
